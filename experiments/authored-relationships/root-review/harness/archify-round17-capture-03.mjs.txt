async function viewport(width=1440,height=900){await page.cdp('Emulation.setDeviceMetricsOverride',{width,height,deviceScaleFactor:1,mobile:false});await page.cdp('Emulation.setEmulatedMedia',{features:[{name:'prefers-reduced-motion',value:'reduce'},{name:'prefers-color-scheme',value:'light'}]});}
async function state(){return await page.evaluate(()=>{const panel=document.querySelector('#semantic-lens'),p=panel.querySelector('.semantic-lens-instruction');const rect=e=>{const r=e.getBoundingClientRect();return{x:r.x,y:r.y,width:r.width,height:r.height}};return{url:location.href,viewport:[innerWidth,innerHeight],theme:document.documentElement.dataset.theme,scroll:[scrollX,scrollY],title:document.querySelector('#semantic-lens-title').textContent,instruction:p.textContent,status:document.querySelector('#semantic-lens-status').textContent,guideHint:document.querySelector('[data-guide-action="lens"] small').textContent,selectedKinds:[...panel.querySelectorAll('[aria-pressed="true"]')].map(e=>e.dataset.kind).sort(),selectedNodes:[...new Set([...document.querySelectorAll('[data-node-id][data-lens-selected]')].map(e=>e.dataset.nodeId))].sort(),peerNodes:[...new Set([...document.querySelectorAll('[data-node-id][data-lens-peer]')].map(e=>e.dataset.nodeId))].sort(),edges:[...new Set([...document.querySelectorAll('[data-edge-from][data-edge-to][data-lens-match]')].map(e=>[e.getAttribute('data-edge-key')||'',e.dataset.edgeFrom,e.dataset.edgeTo].join('|')))].sort(),panel:rect(panel),instructionRect:rect(p),panelScroll:[panel.clientWidth,panel.scrollWidth,panel.clientHeight,panel.scrollHeight],pageWidth:[document.documentElement.clientWidth,document.documentElement.scrollWidth],zoom:document.querySelector('[aria-label*="Reset diagram view"]')?.textContent?.trim()||null};});}
async function capture(name,width=1440,height=900,photo=true){await viewport(width,height);await page.evaluate(async()=>{window.scrollTo(0,0);await document.fonts.ready;await new Promise(r=>requestAnimationFrame(()=>requestAnimationFrame(r)));});const data=await state();if(data.viewport[0]!==width||data.viewport[1]!==height)throw Error('Viewport mismatch '+name);const record={name,...data};if(photo){record.image=name+'.png';await page.screenshot({path:out+'/'+record.image});}records.push(record);await fs.writeFile(out+'/observations.json',JSON.stringify(records,null,2)+'\n');console.log('recorded',name,data.viewport,data.status);}
async function open(url){await page.goto(url);await viewport();await page.waitForSelector('#btn-semantic-lens');await fs.writeFile(out+'/last-navigation.snapshot.txt',String(await page.snapshot()));await page.click('loc=css:#btn-semantic-lens',{label:'Open diagram relationship lens'});await page.waitForSelector('#semantic-lens',{state:'visible'});}
async function choose(kind){await page.click('loc=css:#semantic-lens-kinds button[data-kind="'+kind+'"]',{label:'Select diagram node type'});await page.waitForFunction(k=>document.querySelector('#semantic-lens-kinds button[data-kind="'+k+'"]').getAttribute('aria-pressed')==='true',kind);}
async function guide(name){await page.click('loc=css:#semantic-lens-close',{label:'Close relationship lens'});await page.click('loc=css:button[aria-label="Open diagram guide"]',{label:'Read diagram guide hint'});await page.waitForSelector('#diagram-guide',{state:'visible'});await viewport();await page.evaluate(async()=>{window.scrollTo(0,0);await new Promise(r=>requestAnimationFrame(()=>requestAnimationFrame(r)));});const data=await page.evaluate(()=>({viewport:[innerWidth,innerHeight],text:document.querySelector('[data-guide-action="lens"] small').textContent,visible:!document.querySelector('#diagram-guide').hidden,theme:document.documentElement.dataset.theme}));await page.screenshot({path:out+'/'+name+'.png'});records.push({name,image:name+'.png',...data});await fs.writeFile(out+'/observations.json',JSON.stringify(records,null,2)+'\n');console.log('recorded',name);}
