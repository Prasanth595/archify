const fs=await import('node:fs/promises');
const task=await taskSpace(24),page=task.page('p1');
const root='/tmp/archify-round18-panel',out=root+'/ui';await fs.mkdir(out,{recursive:true});
await fs.writeFile(out+'/browser-version.json',JSON.stringify(await page.cdp('Browser.getVersion'),null,2)+'\n');
const observations=[];
async function viewport(){await page.cdp('Emulation.setDeviceMetricsOverride',{width:1440,height:900,deviceScaleFactor:1,mobile:false});await page.cdp('Emulation.setEmulatedMedia',{features:[{name:'prefers-reduced-motion',value:'reduce'},{name:'prefers-color-scheme',value:'light'}]});}
async function measure(name,photo=true){await viewport();await page.evaluate(async()=>{await document.fonts.ready;await new Promise(r=>requestAnimationFrame(()=>requestAnimationFrame(r)));});const state=await page.evaluate(()=>{const panel=document.querySelector('#semantic-lens'),container=document.querySelector('.diagram-container');const rect=e=>{const r=e.getBoundingClientRect();return{x:r.x,y:r.y,width:r.width,height:r.height,top:r.top,bottom:r.bottom,left:r.left,right:r.right}};const cr=rect(container),cs=getComputedStyle(container);const bounds={top:Math.max(0,cr.top+container.clientTop),bottom:Math.min(innerHeight,cr.bottom-parseFloat(cs.borderBottomWidth)),left:Math.max(0,cr.left+container.clientLeft),right:Math.min(innerWidth,cr.right-parseFloat(cs.borderRightWidth))};return{url:location.href,viewport:[innerWidth,innerHeight],scroll:[scrollX,scrollY],bodyBackground:getComputedStyle(document.body).backgroundColor,container:cr,containerOverflow:[cs.overflowX,cs.overflowY],visibleBounds:bounds,panel:rect(panel),panelOverflow:getComputedStyle(panel).overflowY,status:document.querySelector('#semantic-lens-status').textContent,selected:[...new Set([...document.querySelectorAll('[data-node-id][data-lens-selected]')].map(e=>e.dataset.nodeId))].sort(),edges:[...new Set([...document.querySelectorAll('[data-edge-from][data-edge-to][data-lens-match]')].map(e=>[e.dataset.edgeFrom,e.dataset.edgeTo].join('|')))].sort(),controls:[...panel.querySelectorAll('button')].map(e=>{const r=rect(e),x=r.left+r.width/2,y=r.top+r.height/2,hit=document.elementFromPoint(x,y);return{id:e.id,kind:e.dataset.kind||null,label:e.getAttribute('aria-label')||e.textContent.trim(),rect:r,centerHit:!!hit&&(e===hit||e.contains(hit)),withinVisibleBounds:r.top>=bounds.top&&r.bottom<=bounds.bottom&&r.left>=bounds.left&&r.right<=bounds.right}})};});observations.push({name,image:photo?name+'.png':null,...state});if(photo)await page.screenshot({path:out+'/'+name+'.png'});await fs.writeFile(out+'/observations.json',JSON.stringify(observations,null,2)+'\n');console.log(name,state.scroll,state.panel.top,state.panel.bottom,state.controls.map(c=>[c.id||c.kind,c.centerHit]));}
async function open(arm,name){await page.goto('file://'+root+'/'+arm+'/'+name+'.html');await viewport();await page.waitForSelector('#btn-semantic-lens');await page.click('loc=css:#btn-semantic-lens',{label:'Open relationship lens'});await page.waitForSelector('#semantic-lens',{state:'visible'});}
async function close(name){await page.keyboard.press('Escape');const s=await page.evaluate(()=>({hidden:document.querySelector('#semantic-lens').hidden,activeElement:document.activeElement.id}));observations.push({name,...s});await fs.writeFile(out+'/observations.json',JSON.stringify(observations,null,2)+'\n');}
